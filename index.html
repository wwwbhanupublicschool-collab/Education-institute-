Google Sheets Cloud Sync

Connect a free Google Apps Script Web App URL (doPost)

Google Apps Script Backend Code (Extensions → Apps Script) 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);
  var sheetName = data.type === 'marks' ? 'Marks_' + data.exam : 'Attendance';
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Timestamp', 'Date/Exam', 'RollNo', 'Name', 'Class', 'Details', 'Summary']);
  }
  data.records.forEach(function(r) {
    var details = data.type === 'marks'
      ? 'M:' + r.math + ' S:' + r.science + ' E:' + r.english + ' Soc:' + r.social + ' Tel/Hin:' + r.lang
      : r.status;
    var summary = data.type === 'marks' ? r.percentage + '% (' + r.grade + ')' : r.phone;
    sheet.appendRow([new Date(), data.date || data.exam, r.rollNo, r.name, r.className, details, summary]);
  });
  return ContentService.createTextOutput(JSON.stringify({status: 'ok'}));
}
